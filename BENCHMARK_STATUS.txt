【Calm 持久化写入】
pub fn write_and_persist(&self, records: &[TestRecord]) -> (usize, f64) {
    let start = Instant::now();
    // 构建 JSON 数据
    self.partition.upsert_json(&batch_data)
    // 关键：立即持久化到磁盘
    self.partition.flush(true)  ← 确保数据写入磁盘
    // 计算吞吐量（含持久化）
    records.len() as f64 / elapsed.as_secs_f64()
}

【Tantivy 持久化写入】
pub fn write_and_persist(&self, records: &[TestRecord]) -> (usize, f64) {
    let start = Instant::now();
    // 添加所有文档
    for record in records {
        index_writer.add_document(document)
    }
    // 关键：提交索引到磁盘
    index_writer.commit()  ← 确保索引写入磁盘
    // 计算吞吐量（含提交）
    records.len() as f64 / elapsed.as_secs_f64()
}
